<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\Inventory;
use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;
use Carbon\Carbon;

class AdminInventoryController extends Controller
{
    /**
     * Display the Master Inventory dashboard and Proactive Action Center.
     */
    public function index()
    {
        // 1. Fetch the main inventory list with pagination and eager load the assignee
        $inventories = Inventory::with('assignee')->latest()->paginate(15);

        // 2. Calculate Low Stock Alerts (Quantity <= Reorder Level)
        $lowStockAlerts = Inventory::whereNotNull('reorder_level')
            ->whereColumn('quantity', '<=', 'reorder_level')
            ->get();

        // 3. Calculate Expiring / End-of-Life Alerts (Expires within the next 90 days)
        $expiringAlerts = Inventory::whereNotNull('expiry_date')
            ->where('expiry_date', '<=', Carbon::now()->addDays(90))
            ->orderBy('expiry_date', 'asc')
            ->get();

        return view('admin.inventory.index', compact('inventories', 'lowStockAlerts', 'expiringAlerts'));
    }

    /**
     * Show the form for manually creating a new asset.
     */
    public function create()
    {
        // Fetch users to populate the "Assigned To" dropdown (Teachers, Staff, etc.)
        $users = User::orderBy('last_name')->get();
        
        return view('admin.inventory.create', compact('users'));
    }

    /**
     * Store a newly created asset in storage.
     */
    public function store(Request $request)
    {
        $validated = $request->validate([
            'item_name' => 'required|string|max:255',
            'category' => 'required|string|max:255',
            'property_number' => 'nullable|string|unique:inventories,property_number|max:255',
            'lifespan_type' => 'required|in:Non-Expendable,Semi-Expendable,Consumable',
            'quantity' => 'required|integer|min:0',
            'unit' => 'required|string|max:50',
            'condition' => 'required|in:Serviceable,Needs Minor Repair,Needs Major Repair,Unserviceable,Missing',
            'reorder_level' => 'nullable|integer|min:0',
            'assignment_type' => 'required|in:Facility,Personal',
            'location' => 'nullable|string|max:255',
            'assigned_to' => 'nullable|exists:users,id',
            'funding_source' => 'nullable|string|max:255',
            'date_acquired' => 'nullable|date',
            'expiry_date' => 'nullable|date',
            'photo' => 'nullable|image|max:2048', // Max 2MB image
        ]);

        // Handle Photo Upload
        if ($request->hasFile('photo')) {
            $validated['photo_path'] = $request->file('photo')->store('inventory_photos', 'public');
        }

        // Remove the 'photo' key from the array as we use 'photo_path' in the DB
        unset($validated['photo']);

        Inventory::create($validated);

        return redirect()->route('admin.inventory.index')
            ->with('success', 'New asset successfully registered to the Master Inventory.');
    }

    /**
     * Show the form for editing the specified asset.
     */
    public function edit(Inventory $inventory)
    {
        $users = User::orderBy('last_name')->get();
        return view('admin.inventory.edit', compact('inventory', 'users'));
    }

    /**
     * Update the specified asset in storage.
     */
    public function update(Request $request, Inventory $inventory)
    {
        $validated = $request->validate([
            'item_name' => 'required|string|max:255',
            'category' => 'required|string|max:255',
            'property_number' => 'nullable|string|max:255|unique:inventories,property_number,' . $inventory->id,
            'lifespan_type' => 'required|in:Non-Expendable,Semi-Expendable,Consumable',
            'quantity' => 'required|integer|min:0',
            'unit' => 'required|string|max:50',
            'condition' => 'required|in:Serviceable,Needs Minor Repair,Needs Major Repair,Unserviceable,Missing',
            'reorder_level' => 'nullable|integer|min:0',
            'assignment_type' => 'required|in:Facility,Personal',
            'location' => 'nullable|string|max:255',
            'assigned_to' => 'nullable|exists:users,id',
            'funding_source' => 'nullable|string|max:255',
            'date_acquired' => 'nullable|date',
            'expiry_date' => 'nullable|date',
            'photo' => 'nullable|image|max:2048',
        ]);

        // Handle Photo Upload & Deletion of old photo
        if ($request->hasFile('photo')) {
            if ($inventory->photo_path) {
                Storage::disk('public')->delete($inventory->photo_path);
            }
            $validated['photo_path'] = $request->file('photo')->store('inventory_photos', 'public');
        }
        
        unset($validated['photo']);

        $inventory->update($validated);

        return redirect()->route('admin.inventory.index')
            ->with('success', 'Asset details have been successfully updated.');
    }

    /**
     * Remove the specified asset from storage.
     */
    public function destroy(Inventory $inventory)
    {
        // Delete the photo from storage if it exists
        if ($inventory->photo_path) {
            Storage::disk('public')->delete($inventory->photo_path);
        }

        $inventory->delete();

        return redirect()->route('admin.inventory.index')
            ->with('success', 'Asset has been permanently removed from the registry.');
    }
}

<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\LisStaging;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use App\Imports\Sf1Import;
use Maatwebsite\Excel\Facades\Excel;

class LisSyncController extends Controller
{
    /**
     * Upload page (LIS Sync index)
     */
    public function index()
    {
        // June-based school year default
        $currentMonth = (int) date('n');
        $currentYear  = (int) date('Y');
        $baseStartYear = $currentMonth >= 6 ? $currentYear : $currentYear - 1;

        $defaultSchoolYear = "{$baseStartYear}-" . ($baseStartYear + 1);

        // Rolling years shown in dropdown: previous 3, current, next 1
        $schoolYears = [];
        for ($i = -3; $i <= 1; $i++) {
            $start = $baseStartYear + $i;
            $schoolYears[] = "{$start}-" . ($start + 1);
        }

        $summary = session('sync_summary', [
            'last_run' => LisStaging::max('updated_at'),
            'inserted' => 0,
            'updated'  => 0,
            'skipped'  => 0,
        ]);

        return view('admin.lis_sync.index', compact('schoolYears', 'defaultSchoolYear', 'summary'));
    }

    /**
     * Handle the upload of the official LIS SF1 file using Maatwebsite Excel.
     */
    public function upload(Request $request)
    {
        $request->validate([
            // Added mimes validation to ensure only Excel/CSV files pass through
            'sf1_file' => ['required', 'file', 'mimes:xlsx,xls,csv,txt', 'max:10240'], // Max 10MB
            'school_year' => ['required', 'string'],
        ]);

        try {
            // Initialize our custom SF1 Importer
            $import = new Sf1Import($request->input('school_year'));

            // Let Maatwebsite process the Excel file
            Excel::import($import, $request->file('sf1_file'));

            // Update the summary session with the stats from the import class
            session(['sync_summary' => [
                'last_run' => now(),
                'inserted' => $import->inserted,
                'updated'  => $import->updated,
                'skipped'  => 0,
            ]]);

            return redirect()->route('admin.lis_sync.index')
                             ->with('success', "SF1 Processed Successfully! Added: {$import->inserted}, Updated: {$import->updated}");

        } catch (\Exception $e) {
            logger()->error('SF1 Import Failed: ' . $e->getMessage());
            return back()->withErrors(['sf1_file' => 'Failed to process the file. Please ensure it is a valid SF1 format. Error: ' . $e->getMessage()]);
        }
    }

    /**
     * Manage records list with search, filter, and pagination
     */
    public function manage(Request $request)
    {
        $schoolYears = LisStaging::select('school_year')
            ->distinct()
            ->orderBy('school_year', 'desc')
            ->pluck('school_year');

        // Optional filters
        $search = $request->input('search');
        $syFilter = $request->input('school_year');

        $query = LisStaging::query();

        if ($syFilter) {
            $query->where('school_year', $syFilter);
        }

        if ($search) {
            $query->where(function ($q) use ($search) {
                $q->where('lrn', 'like', "%{$search}%")
                  ->orWhere('last_name', 'like', "%{$search}%")
                  ->orWhere('first_name', 'like', "%{$search}%");
            });
        }

        $records = $query->orderBy('last_name')->orderBy('first_name')->paginate(20)->withQueryString();

        return view('admin.lis_sync.manage', compact('records', 'schoolYears', 'syFilter', 'search'));
    }

    /**
     * Show edit form for a specific staging record.
     */
    public function edit(LisStaging $lisStaging)
    {
        return view('admin.lis_sync.edit', compact('lisStaging'));
    }

    /**
     * Update staging record.
     */
    public function update(Request $request, LisStaging $lisStaging)
    {
        $validated = $request->validate([
            'lrn'            => 'required|string|max:20',
            'last_name'      => 'required|string|max:100',
            'first_name'     => 'required|string|max:100',
            'middle_name'    => 'nullable|string|max:100',
            'extension_name' => 'nullable|string|max:10',
            'gender'         => 'nullable|string|max:20',
            'birthdate'      => 'nullable|string|max:50',
            'grade_level'    => 'required|string|max:50',
            'section'        => 'required|string|max:100',
            'curriculum'     => 'nullable|string|max:100',
            'status'         => 'nullable|string|max:50',
        ]);

        // Prevent updating LRN to an existing one in the same SY
        if (
            $validated['lrn'] !== $lisStaging->lrn
            && LisStaging::where('lrn', $validated['lrn'])
                ->where('school_year', $lisStaging->school_year)
                ->exists()
        ) {
            return back()->withErrors(['lrn' => 'This LRN already exists for this School Year.']);
        }

        $lisStaging->update($validated);

        return redirect()
            ->route('admin.lis_sync.manage', ['school_year' => $lisStaging->school_year])
            ->with('success', 'Record updated successfully.');
    }

    /**
     * Delete a single staging record.
     */
    public function destroy(LisStaging $lisStaging)
    {
        $sy = $lisStaging->school_year;
        $lisStaging->delete();

        return redirect()
            ->route('admin.lis_sync.manage', ['school_year' => $sy])
            ->with('success', 'Record deleted.');
    }

    /**
     * Show absolute reset confirmation form.
     */
    public function resetForm()
    {
        $schoolYears = LisStaging::select('school_year')
            ->distinct()
            ->orderBy('school_year', 'desc')
            ->pluck('school_year');

        return view('admin.lis_sync.reset', compact('schoolYears'));
    }

    /**
     * ABSOLUTE delete all records for a selected SY (type-to-confirm).
     */
    public function resetYear(Request $request)
    {
        $validated = $request->validate([
            'school_year'         => ['required', 'string'],
            'confirm_school_year' => ['required', 'string'],
        ]);

        if ($validated['school_year'] !== $validated['confirm_school_year']) {
            return back()
                ->withErrors(['confirm_school_year' => 'School year confirmation does not match.'])
                ->withInput();
        }

        $sy = $validated['school_year'];

        $deletedCount = DB::transaction(function () use ($sy) {
            $query = LisStaging::query()->where('school_year', $sy);
            $count = (clone $query)->count();

            $builder = (clone $query);

            if (method_exists($builder, 'forceDelete')) {
                $builder->forceDelete();
            } else {
                $builder->delete();
            }

            return $count;
        });

        return redirect()
            ->route('admin.lis_sync.reset')
            ->with('success', "ABSOLUTE DELETE complete: {$deletedCount} record(s) permanently removed for SY {$sy}.");
    }
}

<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Http\Requests\Admin\StoreAssetAssignmentRequest;
use App\Models\Asset;
use App\Models\AssetAssignment;
use App\Models\User;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\View\View;

class MasterInventoryController extends Controller
{
    public function index(): View
    {
        $assets = Asset::with('currentAssignment.user')->latest()->get();
        $teachers = User::where('role', 'teacher')->where('is_active', true)->orderBy('last_name')->get();

        return view('admin.inventory.index', compact('assets', 'teachers'));
    }

    public function store(StoreAssetAssignmentRequest $request): RedirectResponse
    {
        $asset = Asset::updateOrCreate([
            'serial_number' => $request->string('serial_number')->toString(),
        ], [
            'name' => $request->string('name')->toString(),
            'asset_type' => $request->string('asset_type')->toString(),
            'condition_status' => $request->input('condition_status', 'functional'),
            'status' => 'assigned',
            'notes' => $request->input('notes'),
        ]);

        $asset->assignments()->whereNull('released_at')->update([
            'released_at' => now(),
            'assignment_status' => 'released',
        ]);

        AssetAssignment::create([
            'asset_id' => $asset->id,
            'user_id' => $request->integer('user_id'),
            'assigned_by' => $request->user()->id,
            'assigned_at' => now(),
            'assignment_status' => 'pending_signature',
            'notes' => $request->input('notes'),
        ]);

        return redirect()->route('admin.inventory.index')->with('status', 'Asset assignment recorded successfully.');
    }

    public function update(Request $request, Asset $inventory): RedirectResponse
    {
        $data = $request->validate([
            'condition_status' => ['required', 'in:functional,damaged,lost,condemned,replaced'],
        ]);

        $inventory->update($data);

        return redirect()->route('admin.inventory.index')->with('status', 'Asset condition updated successfully.');
    }
}

use Illuminate\Support\Facades\Password;
use Illuminate\Support\Str;
use Illuminate\Validation\Rules;
use Illuminate\Validation\ValidationException;
use Illuminate\View\View;

class NewPasswordController extends Controller
{
    /**
     * Display the password reset view.
     */
    public function create(Request $request): View
    {
        return view('auth.reset-password', ['request' => $request]);
    }

    /**
     * Handle an incoming new password request.
     *
     * @throws ValidationException
     */
    public function store(Request $request): RedirectResponse
    {
        $request->validate([
            'token' => ['required'],
            'email' => ['required', 'email'],
            'password' => ['required', 'confirmed', Rules\Password::defaults()],
        ]);

        // Here we will attempt to reset the user's password. If it is successful we
        // will update the password on an actual user model and persist it to the
        // database. Otherwise we will parse the error and return the response.
        $status = Password::reset(
            $request->only('email', 'password', 'password_confirmation', 'token'),
            function (User $user) use ($request) {
                $user->forceFill([
                    'password' => Hash::make($request->password),
                    'remember_token' => Str::random(60),
                ])->save();

                event(new PasswordReset($user));
            }
        );

        // If the password was successfully reset, we will redirect the user back to
        // the application's home authenticated view. If there is an error we can
        // redirect them back to where they came from with their error message.
        return $status == Password::PASSWORD_RESET
                    ? redirect()->route('login')->with('status', __($status))
                    : back()->withInput($request->only('email'))
                        ->withErrors(['email' => __($status)]);
    }
}

                              ->where('incident_reports.status', '!=', 'overridden');
                        });
                    })

    public function generateBatch(Request $request)
    {
        $validated = $request->validate([
            'school_year' => ['required', 'string'],
            'scope' => ['required', 'in:grade,all'],
            'grade_level' => ['nullable', 'string'],
        ]);

        $schoolYear = trim($validated['school_year']);
        $scope = $validated['scope'];
        $gradeLevel = trim($validated['grade_level'] ?? '');
        $gradeLevels = collect(['7', '8', '9', '10', '11', '12']);

        if ($scope === 'grade' && $gradeLevel === '') abort(422, 'Grade required.');

        $base = DB::table('lis_stagings as ls')->where('ls.school_year', $schoolYear);

        if ($scope === 'grade') {
            $base->where(function($q) use ($gradeLevel) {
                $q->where('ls.grade_level', $gradeLevel)
                  ->orWhere('ls.grade_level', 'Grade ' . $gradeLevel)
                  ->orWhere('ls.grade_level', '0' . $gradeLevel);
            });
        } else {
            $base->where(function($q) use ($gradeLevels) {
                $q->whereIn('ls.grade_level', $gradeLevels->all())
                  ->orWhereIn('ls.grade_level', $gradeLevels->map(fn($g) => "Grade $g")->all())
                  ->orWhereIn('ls.grade_level', $gradeLevels->map(fn($g) => "0$g")->all());
            });
        }

        $rows = (clone $base)
            ->leftJoin('students as s', 's.lrn', '=', 'ls.lrn')
            ->leftJoin('incident_reports as ir', function ($join) {
                $join->on('ir.student_id', '=', 's.id')
                    ->where(function ($q) {
                        $q->where(function ($sub) {
                            $sub->where('ir.severity', 'major')
                                ->whereNotIn('ir.status', ['resolved', 'overridden']);
                        })->orWhere(function ($sub) {
                            $sub->where('ir.severity', 'critical')
                                ->where('ir.status', '!=', 'overridden');
                        });
                    });
            })
            // Added ls.id to ensure the certificate gets a unique ID if available!
            ->groupBy(
                'ls.id',
                'ls.lrn',
                'ls.first_name',
                'ls.last_name',
                'ls.middle_name',
                'ls.extension_name',
                'ls.grade_level',
                'ls.section',
                'ls.school_year'
            )
            ->selectRaw("
                ls.id,
                ls.lrn,
                ls.first_name,
                ls.last_name,
                ls.middle_name,
                ls.extension_name,
                ls.grade_level,
                ls.section,
                ls.school_year,
                COUNT(ir.id) as blocking_offense_count
            ")
            ->orderBy('ls.last_name')
            ->orderBy('ls.first_name')
            ->get();

        $eligible = $rows->filter(fn ($r) => (int) $r->blocking_offense_count === 0)->values();

        foreach($eligible as $student) {
            $student->full_name = trim(($student->first_name ?? '') . ' ' . ($student->last_name ?? ''));
        }

        $content = view('counselor.good_moral.batch', [
            'students' => $eligible,
            'schoolYear' => $schoolYear,
            'scope' => $scope,
            'gradeLevel' => $gradeLevel,
        ])->render();

        $fileBase = $scope === 'all' ? "good-moral-eligible-SY-{$schoolYear}-all-grades" : "good-moral-eligible-SY-{$schoolYear}-grade-{$gradeLevel}";

        return response()->streamDownload(function () use ($content) {
            echo $content;
        }, Str::slug($fileBase) . '.html', ['Content-Type' => 'text/html; charset=UTF-8']);
    }

    public function store(Request $request): RedirectResponse
    {
        // 1. Validate for an array of LRNs and the new 'other' intervention field
        $request->validate([
            'lrns'              => 'required|array|min:1',
            'lrns.*'            => 'required|string',
            'offense_type'      => 'required|string|max:255',
            'severity'          => 'required|in:minor,major,critical',
            'narrative'         => 'required|string',
            'incident_date'     => 'required|date',
            'intervention_type' => 'required|string',
            'intervention_custom'=> 'required_if:intervention_type,other|nullable|string|max:255',
            'status'            => 'required|in:in_intervention,resolved,escalated,turned_over,pending',
            'scheduled_at'      => 'nullable|date',
            'notes'             => 'nullable|string',
            'evidence_photo'    => 'nullable|image|mimes:jpeg,png,jpg,gif|max:2048',
            'redirect_to'       => 'nullable|string',
        ]);

        $path = null;
        if ($request->hasFile('evidence_photo')) {
            $path = $request->file('evidence_photo')->store('incident_evidence', 'public');
        }

        $finalIntervention = $request->intervention_type === 'other'
            ? $request->intervention_custom
            : $request->intervention_type;

        $isGroup = count($request->lrns) > 1;
        $finalNarrative = $isGroup
            ? "[GROUP INCIDENT] " . $request->narrative
            : $request->narrative;

        $firstStudentId = null;

        DB::transaction(function () use ($request, $path, $finalIntervention, $finalNarrative, &$firstStudentId) {

            foreach ($request->lrns as $index => $lrn) {
                $stagedStudent = DB::table('lis_stagings')->where('lrn', $lrn)->first();

                if (!$stagedStudent) {
                    abort(404, "Student with LRN {$lrn} not found in LIS records.");
                }

                $student = Student::firstOrCreate(
                    ['lrn' => $stagedStudent->lrn],
                    [
                        'first_name'  => $stagedStudent->first_name ?? 'Unknown',
                        'last_name'   => $stagedStudent->last_name ?? 'Unknown',
                        'middle_name' => $stagedStudent->middle_name ?? null,
                        'grade_level' => $stagedStudent->grade_level ?? null,
                        'section'     => $stagedStudent->section ?? null,
                    ]
                );

                if ($index === 0) {
                    $firstStudentId = $student->id;
                }

                IncidentReport::create([
                    'student_id'        => $student->id,
                    'teacher_id'        => auth()->id(),
                    'offense_type'      => $request->offense_type,
                    'severity'          => $request->severity,
                    'narrative'         => $finalNarrative,
                    'incident_date'     => $request->incident_date,
                    'status'            => $request->status,
                    'intervention_type' => $finalIntervention,
                    'scheduled_at'      => $request->scheduled_at,
                    'counselor_notes'   => $request->notes,
                    'evidence_path'     => $path,
                ]);
            }
        });

        if ($request->filled('redirect_to')) {
            return redirect($request->string('redirect_to')->toString())
                ->with('status', 'Behavioral incident successfully logged.');
        }

        if (count($request->lrns) === 1 && $firstStudentId) {
            return redirect()->route('counselor.violators.show', $firstStudentId)
                ->with('status', 'Behavioral incident successfully logged.');
        }

        return redirect()->route('counselor.incidents.index')
            ->with('status', 'Group behavioral incident successfully logged.');
    }

            ->orderBy('last_name')
            ->orderBy('first_name')

    public function scan(Request $request)
    {
        $request->validate([
            'cot_image' => 'required|image|mimes:jpeg,png,jpg|max:5120',
        ]);

        $imagePath =$request->file('cot_image')->store('temp_scans');
        $absolutePath = Storage::path($imagePath);

        try {
            $apiKey = env('GEMINI_API_KEY');
            if (!$apiKey) {

            // Convert image to Base64 for the Gemini API
            $mimeType = mime_content_type($absolutePath);
            $base64Image = base64_encode(file_get_contents($absolutePath));

            $prompt = "You are an expert AI assistant tasked with extracting data from a Philippine DepEd Classroom Observation Tool (COT) rating sheet.
            Carefully read the handwritten text on this form.
            Extract the observed teacher's full name and the numerical scores (ranging from 2 to 8) for Indicators 1 through 9.
            Return ONLY a valid, raw JSON object without any markdown wrapping (do not use ```json).
            Format exactly like this:
            {

                    \"4\": 7,
                    \"5\": 5,
                    \"6\": 6,
                    \"7\": 7,
                    \"8\": 8,
                    \"9\": 4
                }
            }
            If a score is missing, blank, or you cannot read it due to handwriting, set the value to null.";

            $baseUrl = "https://" . "generativelanguage.googleapis.com";
            $endpoint = "/v1beta/models/gemini-2.5-flash:generateContent?key=";
            $apiUrl = $baseUrl . $endpoint . trim($apiKey);

            $response = Http::withOptions(['verify' => false])
                ->withHeaders(['Content-Type' => 'application/json'])
                ->timeout(60)
                ->post($apiUrl, [
                    'contents' => [
                        [
                            'parts' => [
                                ['text' => $prompt],
                                [
                                    'inlineData' => [
                                        'mimeType' => $mimeType,
                                        'data' => $base64Image
                                    ]
                                ]
                            ]
                        ]
                    ],
                    'generationConfig' => [
                        'temperature' => 0.1, // Low temperature for factual extraction
                        'responseMimeType' => 'application/json',
                    ]
                ]);

            // Clean up the temp file immediately after API call
            unlink($absolutePath);

            if (!$response->successful()) {
                throw new \Exception('Google API Error: ' . $response->body());
            }

            $result = $response->json();
            $aiText = $result['candidates'][0]['content']['parts'][0]['text'] ?? '{}';

            // Parse the JSON returned by Gemini
            $extractedData = json_decode(trim($aiText), true);

            if (!$extractedData || !isset($extractedData['scores'])) {
                 return back()->with('error', 'AI Scanner failed to return structured data. Please ensure the image is clear.');
            }

            // Map the JSON to our expected session array (strictly indicators 1-9)
            $scannedScores = [];
            $missingCount = 0;

            for ($i = 1; $i <= 9; $i++) {
                $score = $extractedData['scores'][(string)$i] ?? null;

                // Validate score constraints (2 to 8)
                if (is_numeric($score) && $score >= 2 && $score <= 8) {
                    $scannedScores[$i] = (int)$score;
                } else {
                    $scannedScores[$i] = '';
                    $missingCount++;
                }
            }

            session()->put('scanned_scores', $scannedScores);

            // --- TEACHER NAME DETECTION ---
            $matchedTeacher = null;
            $extractedName = strtolower(trim($extractedData['teacher_name'] ?? ''));

            if (!empty($extractedName)) {
                $teachers = User::whereHas('roles', function ($query) {
                    $query->where('name', 'teacher');
                })->get();

                $highestSimilarity = 0;

                foreach ($teachers as $teacher) {
                    $firstName = strtolower(trim($teacher->first_name));
                    $lastName = strtolower(trim($teacher->last_name));

                    // Compare both First Last and Last First formats
                    $format1 = $firstName . ' ' . $lastName;
                    $format2 = $lastName . ' ' . $firstName;

                    similar_text($extractedName, $format1, $percent1);
                    similar_text($extractedName, $format2, $percent2);

                    $maxPercent = max($percent1, $percent2);

                    // 75% threshold accounts for messy handwriting interpretations (e.g., "Jalani" vs "Jelani")
                    if ($maxPercent >= 75 && $maxPercent > $highestSimilarity) {
                        $highestSimilarity = $maxPercent;
                        $matchedTeacher = $teacher;
                    }
                }
            }

            $completenessMsg = $missingCount === 0
                ? "All 9 indicators were successfully extracted."
                : "Warning: {$missingCount} indicators require manual verification.";

            if ($matchedTeacher) {
                return redirect()->route('school_head.cot.create', $matchedTeacher->id)
                    ->with('status', "Form detected for {$matchedTeacher->first_name} {$matchedTeacher->last_name}. {$completenessMsg}");
            } else {
                return redirect()->route('school_head.cot.index')
                    ->with('show_teacher_modal', true)
                    ->with('error', "Scores extracted, but the AI could not definitively match the handwriting to a Teacher's name in the database. {$completenessMsg}");
            }

        } catch (\Exception $e) {
            Log::error('Gemini Scanner Error: ' . $e->getMessage());
            if (file_exists($absolutePath)) {
                unlink($absolutePath);
            }
            return back()->with('error', 'AI Scanner failed: ' . $e->getMessage());

        $totalAssets = Inventory::count();

        // 2. Infrastructure Health Index Calculations
        $serviceableCount = Inventory::where('condition', 'Serviceable')->count();
        $repairCount = Inventory::whereIn('condition', ['Needs Minor Repair', 'Needs Major Repair'])->count();
        $unserviceableCount = Inventory::where('condition', 'Unserviceable')->count();

        // Calculate percentages safely (avoiding division by zero)
        $healthPercentage = $totalAssets > 0 ? round(($serviceableCount / $totalAssets) * 100) : 0;
        $repairPercentage = $totalAssets > 0 ? round(($repairCount / $totalAssets) * 100) : 0;
        $unserviceablePercentage = $totalAssets > 0 ? round(($unserviceableCount / $totalAssets) * 100) : 0;

        // 3. Accountability Flags
        $missingCount = Inventory::where('condition', 'Missing')->count();

        // 4. Pending Condemnation & Disposal Queue
        // Fetch all non-expendable/semi-expendable items that the Admin marked as Unserviceable
        $pendingDisposals = Inventory::where('condition', 'Unserviceable')
            ->whereIn('lifespan_type', ['Non-Expendable', 'Semi-Expendable'])
            ->latest()
            ->get();

        // 5. Supply Burn Rate Insights
        // Fetch Consumables that have dropped below their reorder threshold
        $highConsumptionItems = Inventory::where('lifespan_type', 'Consumable')
            ->whereNotNull('reorder_level')
            ->whereColumn('quantity', '<=', 'reorder_level')
            ->orderBy('quantity', 'asc')
            ->take(6) // Limit to top 6 most critical items for the UI card
            ->get();

        return view('school_head.inventory.index', compact(
            'totalAssets',
            'healthPercentage',
            'repairPercentage',
            'unserviceablePercentage',
            'missingCount',
            'pendingDisposals',
            'highConsumptionItems'
        ));
    }

    /**
     * Approve the write-off/disposal of a condemned asset.
     */
    public function approveDisposal(Request $request, $id)
    {
        $asset = Inventory::findOrFail($id);

        // Ensure only unserviceable items are disposed
        if ($asset->condition !== 'Unserviceable') {
            return back()->with('error', 'Only unserviceable items can be written off.');
        }

        // Action: In a real scenario, you might change status to 'Disposed'
        // or utilize SoftDeletes to remove it from active inventory but keep the record.
        $asset->delete(); // Assuming SoftDeletes is on the Inventory model

        // Log the executive action in the transactions table
        $asset->transactions()->create([
            'user_id' => auth()->id(), // School Head
            'action_type' => 'Condition Change',
            'quantity' => $asset->quantity,
            'status' => 'Completed',
            'notes' => 'Executive Approval: Asset officially condemned and written off the registry.'
        ]);

        return back()->with('success', 'Asset successfully approved for disposal and removed from active registry.');
    }

     * Reject the disposal request (e.g., mandate a repair attempt instead).
     */
    public function rejectDisposal(Request $request, $id)

        $asset->update([

        ]);

            'notes' => 'Executive Rejection: Disposal denied. Flagged for mandatory repair attempt.'
        ]);

        return back()->with('success', 'Disposal rejected. The asset has been flagged for repair.');
    }

            'action_type'  => 'Condition Change',
            'quantity'     => 1,
            'status'       => 'Completed',
            'notes'        => $validated['notes'],

            ['key' => $key],
            ['value' => $value]
        );
    }
}

<?php

namespace App\Services\Ai;

use Illuminate\Support\Facades\Http;
use Throwable;

class DllPrecheckerClient
{
    private const MIN_FALLBACK_SCORE = 0.2;

    public function precheck(array $payload): array
    {
        $external = $this->callExternalService($payload);

        if ($external !== null) {
            return $external;
        }

        return $this->localFallback($payload);
    }

    protected function callExternalService(array $payload): ?array
    {
        $url = config('services.ai_prechecker.url');

        if (! $url) {
            return null;
        }

        try {
            $response = Http::timeout((int) config('services.ai_prechecker.timeout', 10))
                ->acceptJson()
                ->withToken((string) config('services.ai_prechecker.token'))
                ->post(rtrim($url, '/').'/v1/dll-precheck', [
                    'subject' => $payload['subject'] ?? null,
                    'objectives' => $payload['objectives'] ?? null,
                    'content' => $payload['content'] ?? null,
                    'learning_resources' => $payload['learning_resources'] ?? null,
                    'procedures' => $payload['procedures'] ?? null,
                    'reflection' => $payload['reflection'] ?? null,
                ]);

            if (! $response->successful()) {
                return null;
            }

            return $response->json();
        } catch (Throwable) {
            return null;
        }
    }

    protected function localFallback(array $payload): array
    {
        $required = ['subject', 'objectives', 'content', 'learning_resources', 'procedures'];
        $missing = [];

        foreach ($required as $field) {
            if (blank($payload[$field] ?? null)) {
                $missing[] = $field;
            }
        }

        $passed = $missing === [];

        $requiredCount = max(1, count($required));

        return [
            'passed' => $passed,
            'score' => $passed ? 1.0 : max(self::MIN_FALLBACK_SCORE, 1 - (count($missing) / $requiredCount)),
            'provider' => 'local-fallback',
            'feedback' => $passed
                ? ['DLL draft is complete enough to submit for review.']
                : ['Please complete the following sections: '.implode(', ', $missing).'.'],
        ];
    }
}

<?php

namespace App\Services;

class TrackEdCoTeacher
{
    /**
     * Generates a highly contextual pedagogical suggestion based on the subject and the teacher's goal.
     */
    public function generateSuggestion(string $subject, string $intentions, string $type): string
    {
        $subject = strtolower(trim($subject));

        // 1. AI "Thinking" Step: Extract the core topic
        $topic = $this->extractCoreTopic($intentions);

        // 2. AI "Thinking" Step: Detect cognitive demand
        $cognitiveLevel = $this->detectCognitiveLevel($intentions);

        // 3. Route to the appropriate dynamic generator
        if ($type === 'learning_experience') {
            return $this->buildActivity($subject, $topic, $cognitiveLevel);
        }

        return $this->buildAssessment($subject, $topic, $cognitiveLevel);
    }

    /**
     * NLP Heuristic: Strips pedagogical filler words to find the actual subject matter.
     * Example: "To understand the process of photosynthesis" -> "process of photosynthesis"
     */
    protected function extractCoreTopic(string $intentions): string
    {
        $clean = strtolower($intentions);

        // Phrases to aggressively remove
        $fillerPhrases = [
            '/^the (learner|learners|student|students) will (be able to )?/i',
            '/^(to )?(understand|learn|know|comprehend|identify|discuss|analyze|evaluate|create|describe) /i',
            '/\b(about|how to|the concept of)\b/i'
        ];

        foreach ($fillerPhrases as $pattern) {
            $clean = preg_replace($pattern, '', $clean);
        }

        return trim($clean) ?: 'the core concepts'; // Fallback if they only typed a single word
    }

    /**
     * Maps the teacher's verbs to Bloom's Taxonomy to adjust the difficulty of the suggestion.
     */
    protected function detectCognitiveLevel(string $intentions): string
    {
        $intentions = strtolower($intentions);

        $hots = ['create', 'design', 'evaluate', 'judge', 'analyze', 'compare', 'formulate'];
        $lots = ['define', 'list', 'memorize', 'identify', 'state', 'name'];

        foreach ($hots as $verb) { if (str_contains($intentions, $verb)) return 'HOTS'; }
        foreach ($lots as $verb) { if (str_contains($intentions, $verb)) return 'LOTS'; }

        return 'MOTS'; // Default to application/understanding
    }

    /**
     * ==========================================
     * DYNAMIC ACTIVITY BUILDERS
     * ==========================================
     */
    protected function buildActivity(string $subject, string $topic, string $level): string
    {
        // Randomize groupings so the AI feels dynamic if clicked multiple times
        $groupings = ['in pairs', 'in small triads', 'in groups of four', 'collaboratively'];
        $group = $groupings[array_rand($groupings)];

        if (str_contains($subject, 'math')) {
            if ($level === 'HOTS') {
                return "CPA Approach (Abstract): Present an open-ended, real-world scenario involving {$topic}. Have students work {$group} to formulate multiple mathematical pathways to solve it, then defend their chosen method to the class.";
            }
            return "Guided Practice (I Do, We Do, You Do): Demonstrate the step-by-step calculation for {$topic} on the board. Next, solve a similar problem {$group}. Finally, have students attempt one independently.";
        }

        if (str_contains($subject, 'science')) {
            if ($level === 'LOTS') {
                return "5E Model (Engage): Start with a discrepant event or a short video clip related to {$topic}. Ask students to write down 2 observations and 1 question they have before starting the formal lecture.";
            }
            return "5E Model (Explore & Explain): Set up learning stations. Have students work {$group} to manipulate models or data related to {$topic}. Ask them to construct their own explanations before you provide the scientific definitions.";
        }

        if (str_contains($subject, 'english') || str_contains($subject, 'filipino')) {
            return "Contextualized Application: Provide a localized reading text or visual prompt. Have students work {$group} to analyze how {$topic} is applied in the material, then task them with producing a short, original output using the same structure.";
        }

        if (str_contains($subject, 'values') || str_contains($subject, 'esp') || str_contains($subject, 'ap') || str_contains($subject, 'araling panlipunan')) {
            return "4As Framework (Activity & Analysis): Initiate a short role-play or present a localized dilemma regarding {$topic}. <?php

namespace App\Services;

use App\Models\DailyLessonLog;
use Illuminate\Support\Collection;

/**
 * TrackEdSmartEngine
 *
 * Rule-based, fully self-contained analysis engine for DepEd Daily Lesson Logs (DLL).
 * Now includes Context-Awareness (Plagiarism detection) and Instructional Alignment.
 */
class TrackEdSmartEngine
{
    /** Weighting profile for the overall compliance score (should sum to 100). */
    protected array $weights = [
        'completeness'     => 30,
        'assessment'       => 20,
        'reflection'       => 15,
        'cognitive_demand' => 20,
        'originality'      => 15, // NEW: Weight for not copy-pasting
    ];

    /** Revised Bloom's / Anderson taxonomy verbs, grouped by cognitive level. */
    protected array $taxonomy = [
        'remembering'   => ['define', 'list', 'recall', 'name', 'state', 'identify', 'label', 'memorize'],
        'understanding' => ['describe', 'explain', 'discuss', 'summarize', 'classify', 'interpret', 'paraphrase'],
        'applying'      => ['apply', 'demonstrate', 'illustrate', 'solve', 'use', 'compute', 'implement'],
        'analyzing'     => ['analyze', 'compare', 'contrast', 'differentiate', 'examine', 'organize', 'distinguish'],
        'evaluating'    => ['evaluate', 'judge', 'critique', 'justify', 'argue', 'assess', 'defend', 'appraise'],
        'creating'      => ['create', 'design', 'formulate', 'construct', 'compose', 'devise', 'plan', 'generate', 'produce'],
    ];

    protected array $lotsLevels = ['remembering', 'understanding'];
    protected array $motsLevels = ['applying', 'analyzing'];
    protected array $hotsLevels = ['evaluating', 'creating'];

    public function analyzeLessonPlan(DailyLessonLog $dll): array
    {
        $dll->load(['entries', 'author']);
        $entries   = $dll->entries;
        $totalDays = $entries->count();

        if ($totalDays === 0) {
            return $this->blankMatrixResponse();
        }

        $cognitive    = $this->scanCognitiveDemand($dll);
        $completeness = $this->scanCompleteness($entries);
        $history      = $this->checkSystemHistory($dll, $entries); // NEW: Contextual Check

        $perDay       = $this->buildPerDayBreakdown($entries);
        $score        = $this->computeComplianceScore($completeness, $cognitive, $history, $totalDays);

        return [
            'brief'                 => $this->buildBrief($totalDays, $cognitive, $completeness, $history, $score),
            'suggested_feedback'    => $this->buildFeedback($completeness, $cognitive, $history, $totalDays),
            'compliance_score'      => $score,
            'rating'                => $this->scoreToRating($score),
            'cognitive_profile'     => $cognitive,
            'completeness'          => $completeness,
            'strengths'             => $this->identifyStrengths($completeness, $cognitive, $history),
            'areas_for_improvement' => $this->identifyImprovements($completeness, $cognitive, $history),
            'per_day'               => $perDay,
        ];
    }

    protected function blankMatrixResponse(): array
    {
        return [
            'brief'                 => "🚨 Critical: Matrix is completely blank. No instructional days recorded.",
            'suggested_feedback'    => "Please complete your daily procedures before routing for supervision.",
            'compliance_score'      => 0,
            'rating'                => 'Not Ratable',
            'cognitive_profile'     => null,
            'completeness'          => null,
            'strengths'             => [],
            'areas_for_improvement' => ['No entries to evaluate.'],
            'per_day'               => [],
        ];
    }

    protected function scanCognitiveDemand(DailyLessonLog $dll): array
    {
        $text = strtolower(trim(
            ($dll->competencies ?? '') . ' ' .
            ($dll->content_standard ?? '') . ' ' .
            ($dll->performance_standard ?? '')
        ));

        $hits = [];
        foreach ($this->taxonomy as $level => $verbs) {
            $hits[$level] = array_values(array_filter($verbs, fn ($verb) => str_contains($text, $verb)));
        }

        $lotsCount = array_sum(array_map(fn ($l) => count($hits[$l]), $this->lotsLevels));
        $motsCount = array_sum(array_map(fn ($l) => count($hits[$l]), $this->motsLevels));
        $hotsCount = array_sum(array_map(fn ($l) => count($hits[$l]), $this->hotsLevels));

        $verdict = match (true) {
            $hotsCount > 0 && $hotsCount >= $lotsCount => 'hots',
            $motsCount > 0 && $motsCount >= $lotsCount => 'mots',
            $lotsCount > 0                              => 'lots',
            default                                     => 'undetermined',
        };

        return [
            'verdict'    => $verdict,
            'hits'       => $hits,
            'lots_count' => $lotsCount,
            'mots_count' => $motsCount,
            'hots_count' => $hotsCount,
        ];
    }

    protected function scanCompleteness(Collection $entries): array
    {
        $missingAssessments = 0;
        $missingReflections = 0;
        $missingActivities  = 0;

        foreach ($entries as $entry) {
            if ($this->isBlank($entry->procedure_e ?? null)) $missingAssessments++;
            if ($this->isBlank($entry->reflection_notes ?? null)) $missingReflections++;
            if ($this->isBlank($entry->procedure_a ?? null)) $missingActivities++;
        }

        return [
            'missing_assessments' => $missingAssessments,
            'missing_reflections' => $missingReflections,
            'missing_activities'  => $missingActivities,
        ];
    }

    /** NEW: Checks the database for copy-pasting from last week */
    protected function checkSystemHistory(DailyLessonLog $dll, Collection $entries): array
    {
        $currentActivitiesText = "";
        foreach ($entries as $entry) {
            $currentActivitiesText .= strtolower($entry->procedure_a . " ");
        }

        $lastLog = DailyLessonLog::with('entries')
            ->where('author_id', $dll->author_id)
            ->where('id', '!=', $dll->id)
            ->latest('created_at')
            ->first();

        $isCopyPasted = false;

        if ($lastLog && !empty(trim($currentActivitiesText))) {
            $lastActivitiesText = "";
            foreach ($lastLog->entries as $entry) {
                $lastActivitiesText .= strtolower($entry->procedure_a . " ");
            }

            if (!empty(trim($lastActivitiesText))) {
                similar_text($currentActivitiesText, $lastActivitiesText, $percentSimilarity);
                if ($percentSimilarity > 85) {
                    $isCopyPasted = true;
                }
            }
        }

        return [
            'is_copy_pasted' => $isCopyPasted
        ];
    }

    protected function buildPerDayBreakdown(Collection $entries): array
    {
        return $entries->values()->map(function ($entry, $i) {
            $issues = [];
            if ($this->isBlank($entry->procedure_a ?? null)) $issues[] = 'No instructional activities logged.';
            if ($this->isBlank($entry->procedure_e ?? null)) $issues[] = 'No formative assessment logged.';
            if ($this->isBlank($entry->reflection_notes ?? null)) $issues[] = 'No reflection notes.';

            return [
                'day'    => $i + 1,
                'date'   => $entry->date ?? null,
                'status' => empty($issues) ? 'complete' : 'incomplete',
                'issues' => $issues,
            ];
        })->toArray();
    }

    protected function computeComplianceScore(array $completeness, array $cognitive, array $history, int $totalDays): int
    {
        $activityRate     = 1 - ($completeness['missing_activities'] / $totalDays);
        $assessmentRate   = 1 - ($completeness['missing_assessments'] / $totalDays);
        $reflectionRate   = 1 - ($completeness['missing_reflections'] / $totalDays);

        $completenessRate = ($activityRate + $assessmentRate + $reflectionRate) / 3;

        $cognitiveScore = match ($cognitive['verdict']) {
            'hots'  => 1.0,
            'mots'  => 0.75,
            'lots'  => 0.5,
            default => 0.3,
        };

        $originalityScore = $history['is_copy_pasted'] ? 0.0 : 1.0;

        $score = ($completenessRate * $this->weights['completeness'])
            + ($assessmentRate * $this->weights['assessment'])
            + ($reflectionRate * $this->weights['reflection'])
            + ($cognitiveScore * $this->weights['cognitive_demand'])
            + ($originalityScore * $this->weights['originality']);

        return (int) round(max(0, min(100, $score)));
    }

    protected function scoreToRating(int $score): string
    {
        return match (true) {
            $score >= 90 => 'Highly Proficient',
            $score >= 75 => 'Proficient',
            $score >= 60 => 'Developing',
            default      => 'Needs Immediate Support',
        };
    }

    protected function buildBrief(int $totalDays, array $cognitive, array $completeness, array $history, int $score): string
    {
        $brief = "🧠 TrackEd Engine Scan: Teacher logged {$totalDays} day(s). ";

        if ($history['is_copy_pasted']) {
            $brief .= "⚠️ Red Flag: Activities are highly identical to the previous week's submission. ";
        }

        $brief .= match ($cognitive['verdict']) {
            'hots'  => "Competencies show strong High-Order Thinking (HOTS) alignment. ",
            'mots'  => "Competencies show moderate-order cognitive demand. ",
            'lots'  => "Competencies lean towards foundational recall (LOTS). ",
            default => "Cognitive level of competencies could not be determined. ",
        };

        if ($completeness['missing_activities'] > 0) {
            $brief .= "{$completeness['missing_activities']} day(s) lack instructional activities. ";
        }
        if ($completeness['missing_assessments'] > 0) {
            $brief .= "{$completeness['missing_assessments']} day(s) lack explicit formative assessments. ";
        }

        return $brief . "Overall compliance score: {$score}/100.";
    }

    protected function buildFeedback(array $completeness, array $cognitive, array $history, int $totalDays): string
    {
        if ($history['is_copy_pasted']) {
            return "Please review your daily procedures. They appear to be a direct duplicate of last week's lesson log. Ensure activities are updated for the current competencies.";
        }

        if ($completeness['missing_activities'] > 0) {
            return "Please detail your step-by-step instructional procedures for the missing day(s).";
        }

        if ($completeness['missing_assessments'] > 0) {
            return "Please specify your formative evaluation methods (e.g., 5-item quiz) for the missing day(s).";
        }

        if ($completeness['missing_reflections'] === $totalDays) {
            return "Reminder: Post-lesson reflections must be completed to finalize the instructional cycle.";
        }

        if ($cognitive['verdict'] === 'lots') {
            return "Consider revising at least one competency per week to require analysis, evaluation, or creation, rather than recall alone.";
        }

        return "Matrix is highly compliant and well-structured. Checked and noted.";
    }

    protected function identifyStrengths(array $completeness, array $cognitive, array $history): array
    {
        $strengths = [];
        if ($completeness['missing_assessments'] === 0) $strengths[] = 'All days have a mapped formative assessment.';
        if ($completeness['missing_reflections'] === 0) $strengths[] = 'Reflection notes are complete for every logged day.';
        if (in_array($cognitive['verdict'], ['hots', 'mots'])) $strengths[] = 'Competencies engage learners above the recall level.';
        if (!$history['is_copy_pasted'] && $completeness['missing_activities'] === 0) $strengths[] = 'Instructional activities are fully mapped and original.';
        return $strengths;
    }

    protected function identifyImprovements(array $completeness, array $cognitive, array $history): array
    {
        $improvements = [];
        if ($history['is_copy_pasted']) {
            $improvements[] = "Ensure instructional activities are unique to the current week's competencies and not duplicated from past logs.";
        }
        if ($completeness['missing_activities'] > 0) {
            $improvements[] = "{$completeness['missing_activities']} day(s) need detailed instructional procedures.";
        }
        if ($completeness['missing_assessments'] > 0) {
            $improvements[] = "{$completeness['missing_assessments']} day(s) need a stated formative assessment.";
        }
        if ($completeness['missing_reflections'] > 0) {
            $improvements[] = "{$completeness['missing_reflections']} day(s) need reflection notes.";
        }
        if (in_array($cognitive['verdict'], ['lots', 'undetermined'])) {
            $improvements[] = 'Raise the cognitive demand of stated competencies (e.g., shift from "define" to "evaluate" or "create").';
        }
        return $improvements;
    }

    protected function isBlank(?string $value): bool
    {
        return trim((string) $value) === '';
    }
}

<?php

namespace App\Http\Requests\Auth;

use Illuminate\Auth\Events\Lockout;
use Illuminate\Foundation\Http\FormRequest;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Validation\ValidationException;
use Illuminate\Support\Str;

class LoginRequest extends FormRequest
{
    public function authorize(): bool
    {
        return true;
    }

    public function rules(): array
    {
        return [
            'login' => ['required', 'string'],
            'password' => ['required', 'string'],
        ];
    }

    public function authenticate(): void
    {
        $this->ensureIsNotRateLimited();

        $login = $this->input('login');

        // Logic: Decide if user used email or employee_id
        $field = filter_var($login, FILTER_VALIDATE_EMAIL) ? 'email' : 'employee_id';

        if (! Auth::attempt([$field => $login, 'password' => $this->password], $this->boolean('remember'))) {
            RateLimiter::hit($this->throttleKey());

            throw ValidationException::withMessages([
                'login' => trans('auth.failed'),
            ]);
        }

        RateLimiter::clear($this->throttleKey());
    }

    public function ensureIsNotRateLimited(): void
    {
        if (! RateLimiter::tooManyAttempts($this->throttleKey(), 5)) {
            return;
        }

        event(new Lockout($this));

        $seconds = RateLimiter::availableIn($this->throttleKey());

        throw ValidationException::withMessages([
            'login' => trans('auth.throttle', [
                'seconds' => $seconds,
                'minutes' => ceil($seconds / 60),
            ]),
        ]);
    }

    public function throttleKey(): string
    {
        return Str::transliterate(Str::lower($this->input('login')).'|'.$this->ip());
    }
}

<?php

namespace App\Imports;

use Illuminate\Support\Collection;
use App\Models\LisStaging;
use Maatwebsite\Excel\Concerns\ToCollection;
use Maatwebsite\Excel\Concerns\WithCalculatedFormulas;

class Sf1Import implements ToCollection, WithCalculatedFormulas
{
    public $inserted = 0;
    public $updated = 0;
    protected $schoolYear;

    // Persist grade and section across the whole file
    protected $currentGradeLevel = 'Unknown';
    protected $currentSection = 'Unknown';

    public function __construct($schoolYear)
    {
        $this->schoolYear = $schoolYear;
    }

    /**
     * @param Collection $collection
     */
    public function collection(Collection $rows)
    {
        foreach ($rows as $row) {

            // 1. COLLAPSE MERGED CELLS (The Fix)
            // Remove all nulls or empty strings, then re-index the array so values are side-by-side
            $cleanedRow = array_values(array_filter($row->toArray(), function($value) {
                return $value !== null && trim((string)$value) !== '';
            }));

            // If the row is completely empty after cleaning, skip it
            if (empty($cleanedRow)) {
                continue;
            }

            // 2. AUTO-DETECT GRADE AND SECTION FROM THE HEADER
            // Because we cleaned the array, the target value is always immediately after the label
            foreach ($cleanedRow as $index => $cellValue) {
                $cellString = trim((string)$cellValue);

                if (stripos($cellString, 'Grade Level') !== false && isset($cleanedRow[$index + 1])) {
                    $this->currentGradeLevel = trim((string)$cleanedRow[$index + 1]);
                }

                if (stripos($cellString, 'Section') !== false && isset($cleanedRow[$index + 1])) {
                    $this->currentSection = trim((string)$cleanedRow[$index + 1]);
                }
            }

            // 3. IDENTIFY STUDENT ROWS
            // After cleaning, Index 0 is LRN, 1 is Name, 2 is Sex, 3 is Birthdate
            $lrn = trim((string)$cleanedRow[0]);

            // Check if the first item is a 12-digit LRN
            if (preg_match('/^\d{12}$/', $lrn)) {

                // Safely grab the data using our predictable indexes
                $rawName   = isset($cleanedRow[1]) ? trim((string)$cleanedRow[1]) : '';
                $gender    = isset($cleanedRow[2]) ? trim((string)$cleanedRow[2]) : '';
                $birthDate = isset($cleanedRow[3]) ? trim((string)$cleanedRow[3]) : '';

                // Standardize Gender (M -> Male, F -> Female)
                if (strtoupper($gender) === 'M') $gender = 'Male';
                if (strtoupper($gender) === 'F') $gender = 'Female';

                // 4. SMART CATCHER: PARSE THE NAME
                $parsedName = $this->parseName($rawName);

                // 5. SAVE TO DATABASE
                $record = LisStaging::updateOrCreate(
                    [
                        'lrn'         => $lrn,
                        'school_year' => $this->schoolYear,
                    ],
                    [
                        'last_name'      => $parsedName['last_name'],
                        'first_name'     => $parsedName['first_name'],
                        'middle_name'    => $parsedName['middle_name'],
                        'extension_name' => $parsedName['extension_name'],
                        'gender'         => $gender,
                        'birthdate'      => $birthDate,
                        'grade_level'    => $this->currentGradeLevel,
                        'section'        => $this->currentSection,
                    ]
                );

                // Track analytics for the controller summary
                if ($record->wasRecentlyCreated) {
                    $this->inserted++;
                } else {
                    $this->updated++;
                }
            }
        }
    }

    /**
     * Parse DepEd Name Format: "LASTNAME, FIRSTNAME, EXTENSION, MIDDLENAME"
     * Handles names with and without extensions.
     */
    private function parseName($rawName)
    {
        // Split the string by comma
        $parts = array_map('trim', explode(',', $rawName));

        $lastName = $parts[0] ?? '';
        $firstName = '';
        $middleName = null;
        $extensionName = null;

        $count = count($parts);

        if ($count === 2) {
            // E.g., "DELA CRUZ, JUAN"
            $firstName = $parts[1];
        } elseif ($count === 3) {
            // E.g., "CAGBAY, MHARK BERNARD, MILLIONES"
            $firstName = $parts[1];
            $middleName = $parts[2];
        } elseif ($count >= 4) {
            // E.g., "CALIBO, ERNIE, JR., CAMINO"
            // The extension is typically the 3rd item, and the middle name is the 4th
            $firstName = $parts[1];
            $extensionName = $parts[2];
            $middleName = $parts[3];
        }

        return [
            'last_name'      => $lastName,
            'first_name'     => $firstName,
            'middle_name'    => $middleName,
            'extension_name' => $extensionName,
        ];
    }
}
